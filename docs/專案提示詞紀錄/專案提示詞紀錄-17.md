## Copilot 回答 管理頁抽象化
可以的！你目前的 `DataTablesPage.tsx` 結構非常清晰，我會幫你抽象成一個通用的頁面元件，命名為：

> **`GenericListPage.tsx`**

這個元件會支援：
- 搜尋欄（可選）
- 新增按鈕（可選）
- 上傳按鈕（可選）
- 檢視模式切換（卡片 / 列表）
- 列表內容由 `GenericListView` 呈現

### ✅ 1. 建立 `GenericListPage.tsx`

```tsx
// src/components/common/GenericListPage.tsx
import {
  Box,
  Grid,
  Typography,
  TextField,
  Button,
  ToggleButtonGroup,
  ToggleButton,
} from "@mui/material";
import GridViewIcon from "@mui/icons-material/GridView";
import FormatListBulletedIcon from "@mui/icons-material/FormatListBulleted";
import AddIcon from "@mui/icons-material/Add";
import UploadIcon from "@mui/icons-material/Upload";
import { useState } from "react";

export interface GenericListPageProps<T> {
  title: string;
  items: T[];
  viewMode?: "card" | "list";
  searchable?: boolean;
  uploadable?: boolean;
  creatable?: boolean;
  searchPlaceholder?: string;
  onSearch?: (keyword: string) => void;
  onCreate?: () => void;
  onUpload?: () => void;
  renderList: (filteredItems: T[], viewMode: "card" | "list") => React.ReactNode;
}

export function GenericListPage<T>({
  title,
  items,
  viewMode = "card",
  searchable = true,
  uploadable = false,
  creatable = false,
  searchPlaceholder = "搜尋",
  onSearch,
  onCreate,
  onUpload,
  renderList,
}: GenericListPageProps<T>) {
  const [searchText, setSearchText] = useState("");
  const [currentViewMode, setCurrentViewMode] = useState(viewMode);

  const handleSearchChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const keyword = e.target.value;
    setSearchText(keyword);
    onSearch?.(keyword);
  };

  const handleViewModeChange = (_: any, mode: "card" | "list") => {
    if (mode) setCurrentViewMode(mode);
  };

  return (
    <Box sx={{ p: 3 }}>
      <Grid container alignItems="center" spacing={2} sx={{ mb: 3 }}>
        <Grid item xs={12} sm={6}>
          <Typography variant="h4" fontWeight="bold">{title}</Typography>
          <ToggleButtonGroup
            value={currentViewMode}
            exclusive
            onChange={handleViewModeChange}
            size="small"
            sx={{ mt: 1 }}
          >
            <ToggleButton value="card"><GridViewIcon /></ToggleButton>
            <ToggleButton value="list"><FormatListBulletedIcon /></ToggleButton>
          </ToggleButtonGroup>
        </Grid>

        <Grid item xs={12} sm={6}>
          <Grid container justifyContent="flex-end" spacing={1}>
            {searchable && (
              <Grid item>
                <TextField
                  label={searchPlaceholder}
                  variant="outlined"
                  size="small"
                  value={searchText}
                  onChange={handleSearchChange}
                />
              </Grid>
            )}
            {creatable && (
              <Grid item>
                <Button variant="contained" startIcon={<AddIcon />} onClick={onCreate}>
                  新增
                </Button>
              </Grid>
            )}
            {uploadable && (
              <Grid item>
                <Button variant="contained" startIcon={<UploadIcon />} onClick={onUpload}>
                  上傳
                </Button>
              </Grid>
            )}
          </Grid>
        </Grid>
      </Grid>

      {/* 列表內容 */}
      {renderList(items, currentViewMode)}
    </Box>
  );
}
```

### ✅ 2. 在 `DataTablesPage.tsx` 中使用

你可以這樣改寫：

```tsx
import { GenericListPage } from "../components/common/GenericListPage";
import { DataTableList } from "../components/DataTablesPage/DataTableList";

export const DataTablesPage = () => {
  const [searchText, setSearchText] = useState("");
  const [uploadDialogOpen, setUploadDialogOpen] = useState(false);
  const [tableInfos, setTableInfos] = useState<DataTableInfo[]>([]);
  const navigate = useNavigate();

  const uploads = useUploadStore((state) => state.uploads);
  const resetUploads = useUploadStore((state) => state.reset);

  useEffect(() => {
    if (uploads.some((u) => u.status === "success")) {
      refreshTableInfos();
    }
  }, [uploads]);

  useEffect(() => {
    refreshTableInfos();
    return () => resetUploads();
  }, [resetUploads]);

  const refreshTableInfos = () => {
    window.api.getAllTableInfos().then(setTableInfos);
  };

  const filteredTables = tableInfos.filter((t) =>
    t.name.toLowerCase().includes(searchText.toLowerCase())
  );

  return (
    <PageWrapper
      breadcrumbItems={[{ label: "資料表格管理", path: "/data-tables" }]}
      content={
        <GenericListPage
          title="資料表格管理"
          items={filteredTables}
          searchable
          creatable
          uploadable
          searchPlaceholder="搜尋表格"
          onSearch={setSearchText}
          onCreate={() => navigate("/data-tables/edit", { state: { editorMode: "create" } })}
          onUpload={() => setUploadDialogOpen(true)}
          renderList={(items, viewMode) => (
            <DataTableList dataTables={items} viewMode={viewMode} refresh={refreshTableInfos} />
          )}
        />
      }
      rightPanelContent={<UploadStatusPanel />}
      rightPanelTitle="上傳狀態"
    />
  );
};
```
