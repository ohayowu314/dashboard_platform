### 1\. 解決同名檔案覆蓋問題

您的觀察非常正確，單純使用檔名作為 ID 在處理同名檔案時會出問題。我們可以為每個上傳任務生成一個唯一的 ID 來解決這個問題。使用 `Date.now()` 和一個隨機數的組合是一個簡單且有效的方法。

#### `src/stores/uploadStore.ts` 修改

```typescript
// src/stores/uploadStore.ts
import { create } from "zustand";
import { devtools } from "zustand/middleware";
import { parseDataFile } from "../utils";
import { DataTableInfo } from "shared/types/dataTable";

export interface FileUploadStatus {
  id: string; // 使用隨機數作為唯一 ID
  fileName: string;
  status: "uploading" | "success" | "failed";
  progress?: number;
  error?: string;
  info?: DataTableInfo;
  timeoutId?: ReturnType<typeof setTimeout>; // 用於儲存定時器 ID
}

interface UploadState {
  uploads: FileUploadStatus[];
  addUpload: (file: File) => string; // 回傳新建立的 ID
  updateUploadStatus: (
    id: string,
    status: FileUploadStatus["status"],
    info?: DataTableInfo,
    error?: string
  ) => void;
  // 處理上傳檔案的非同步邏輯
  startUploads: (files: File[]) => Promise<void>;
  reset: () => void;
  removeUpload: (id: string) => void; // 新增移除方法
}

export const useUploadStore = create<UploadState>()(
  devtools(
    (set, get) => ({
      uploads: [],

      addUpload: (file: File) => {
        const newId = `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
        set((state) => ({
          uploads: [
            ...state.uploads,
            { id: newId, fileName: file.name, status: "uploading" },
          ],
        }));
        return newId;
      },

      updateUploadStatus: (id, status, info, error) => {
        set((state) => ({
          uploads: state.uploads.map((upload) =>
            upload.id === id ? { ...upload, status, info, error } : upload
          ),
        }));
      },

      removeUpload: (id) => {
        set((state) => ({
          uploads: state.uploads.filter((upload) => upload.id !== id),
        }));
      },

      startUploads: async (files: File[]) => {
        const { updateUploadStatus, addUpload, removeUpload } = get();

        for (const file of files) {
          const newId = addUpload(file);
          try {
            const parsedData = await parseDataFile(file);
            const tableInfo = { name: file.name.split(".")[0], description: "" };
            const dataTableInfo = await window.api.uploadTable(
              tableInfo,
              parsedData
            );
            updateUploadStatus(newId, "success", dataTableInfo);
            // 成功後設定定時器
            setTimeout(() => {
              removeUpload(newId);
            }, 10000); // 10 秒後自動移除
          } catch (e: unknown) {
            console.error(`上傳 ${file.name} 失敗:`, e);
            const errorMsg = e instanceof Error ? e.message : "未知錯誤";
            updateUploadStatus(newId, "failed", undefined, errorMsg);
            // 失敗後設定定時器
            setTimeout(() => {
              removeUpload(newId);
            }, 30000); // 30 秒後自動移除
          }
        }
      },
      
      reset: () => {
        set({ uploads: [] });
      },
    }),
    { name: "upload-store" }
  )
);
```

### 總結

1.  **唯一 ID**：我們在 `useUploadStore` 的 `addUpload` 方法中，為每個上傳任務生成了一個唯一的 ID，這徹底解決了同名檔案的覆蓋問題。
2.  **自動移除**：在 `startUploads` 方法中，我們為成功和失敗的任務分別設定了 10 秒和 30 秒的定時器，時間一到，`setTimeout` 會觸發 `removeUpload`，自動從列表中移除該任務。
3.  **手動移除**：在 `UploadStatusPanel` 元件中，我們利用 MUI 的 `ListItem` 的 `secondaryAction` 屬性，為已完成（成功或失敗）的狀態條添加了一個叉叉按鈕，使用者可以隨時手動移除它。

這三個優化使得多檔案上傳功能更加健壯和人性化，提供了更好的使用者體驗。

```ts
export const parseDataFile = (file: File): Promise<DataTableHeaderSchema> => {
  return new Promise((resolve, reject) => {
    if (!file) {
      return reject(new Error("檔案不存在"));
    }
    if (!["text/csv", "application/json"].includes(file.type)) {
      return reject(new Error("不支援的檔案類型"));
    }
    if (file.size > MAX_FILE_SIZE) {
      return reject(new Error("檔案太大"));
    }
    if (file.size === 0) {
      return reject(new Error("檔案為空"));
    }
    if (file.type === "text/csv") {
      // 在 PapaParse 的 .parse 方法中，使用泛型來指定資料型別
      Papa.parse<PapaResultRow>(file, {
        header: true,
        skipEmptyLines: true,
        complete: (results) => {
          if (results.errors.length) {
            console.error(results.errors);
            return reject(new Error("CSV 解析失敗"));
          }
          const headers = results.meta.fields || [];
          // 將每個物件轉換為字串陣列，並確保型別正確
          const rows = results.data.map((row) => Object.values(row));
          resolve({ headers, rows });
        },
      });
    } else if (file.type === "application/json") {
      const reader = new FileReader();
      reader.onload = (event) => {
        try {
          const content = event.target?.result as string;
          const data = JSON.parse(content);

          if (!Array.isArray(data) || data.length === 0) {
            return reject(new Error("JSON 檔案格式不正確，應為陣列且不為空。"));
          }

          const firstItem = data[0];
          if (typeof firstItem !== "object" || firstItem === null) {
            return reject(
              new Error("JSON 檔案內容格式不正確，陣列元素應為物件。")
            );
          }

          const headers = Object.keys(firstItem) as string[];

          // 使用 map 迴圈遍歷每個項目，並透過 headers 陣列的順序取得值
          const rows = data.map((item: { [key: string]: undefined }) => {
            return headers.map((header) => String(item[header]));
          });

          resolve({ headers, rows });
        } catch (error) {
          console.error(error);
          reject(new Error("JSON 檔案解析失敗。"));
        }
      };
      reader.onerror = () => {
        reject(new Error("檔案讀取失敗。"));
      };
      reader.readAsText(file);
    } else {
      reject(new Error("不支援的檔案類型。"));
    }
  });
};
```

### 1. 修改 `parseDataFile` 檢查 JSON 格式

```ts
export const parseDataFile = (file: File): Promise<DataTableHeaderSchema> => {
  return new Promise((resolve, reject) => {
    ...
    if (file.type === "application/json") {
      const reader = new FileReader();
      reader.onload = (event) => {
        try {
          const content = event.target?.result as string;
          const data = JSON.parse(content);

          if (!Array.isArray(data) || data.length === 0) {
            return reject(new Error("JSON 檔案格式不正確，應為陣列且不為空。"));
          }

          const firstItem = data[0];
          if (typeof firstItem !== "object" || firstItem === null || Array.isArray(firstItem)) {
            return reject(
              new Error("JSON 檔案內容格式不正確，陣列元素應為非陣列的物件。")
            );
          }

          const headers = Object.keys(firstItem);

          // 檢查所有物件是否有相同的鍵
          const isConsistent = data.every((item) =>
            headers.every((header) => header in item)
          );
          if (!isConsistent) {
            return reject(
              new Error("JSON 檔案格式不正確，所有物件必須有相同的鍵。")
            );
          }

          const rows = data.map((item) => headers.map((header) => String(item[header])));
          resolve({ headers, rows });
        } catch (error) {
          console.error(error);
          reject(new Error("JSON 檔案解析失敗。"));
        }
      };
      reader.onerror = () => {
        reject(new Error("檔案讀取失敗。"));
      };
      reader.readAsText(file);
    } else {
      reject(new Error("不支援的檔案類型。"));
    }
  });
};
```

作為一位專業的全端桌面軟體程式設計師，我認為處理檔案名稱與現有表格名稱重複的問題，最佳的策略是**讓使用者決定**，並提供一個**清晰且互動性的介面**來預覽和解決衝突。

你提出的三種應對方式各有優缺點，但它們都偏向於**系統單方面決定**，這在資料管理應用中，可能會帶來誤操作的風險。

-----

### 策略分析與建議

我建議的**核心策略**是：**前置檢查 + 互動式選擇 + 批次處理**。

#### 1\. 預先檢查衝突（Pre-flight Check）

在使用者點擊「上傳」之前，讓前端 **`UploadDataTableDialog`** 負責：

1.  **向後端查詢**：發送所選檔案清單，後端 `DataTableAPIHandlers` 應新增一個輕量級 API，快速檢查每個檔名（轉換後的表格名）是否已存在於 `better-sqlite3` 資料庫中。
2.  **標記衝突**：在 `UploadDataTableDialog` 顯示的檔案清單中，用 **MUI Icon**（如 $\text{WarningIcon}$ 或 $\text{ErrorIcon}$）明確標記出衝突的檔案。

#### 2\. 互動式衝突解決（Interactive Conflict Resolution）

當發現衝突時，不要直接開始上傳，而是**暫停**，並在 $\text{UploadDataTableDialog}$ 或一個專門的子組件中，針對每個衝突檔案，提供使用者一個**下拉選單或單選按鈕組**來選擇解決方案。

| 解決方案                         | 專業分析                                                                               | 適用場景                                                         |
| :------------------------------- | :------------------------------------------------------------------------------------- | :--------------------------------------------------------------- |
| **自動更名 (Rename)**            | **👍 最安全、最無損**。保留舊資料，上傳新資料。                                         | **推薦為預設選項**。適用於使用者只是想將檔案視為一個新的資料集。 |
| **更新替換 (Replace/Overwrite)** | **⚠️ 需謹慎**。會永久替換（或刪除重建）現有表格的所有資料和結構。必須搭配**二次確認**。 | 適用於使用者**確定**新檔案是舊表格的最新版本，且不需要舊資料。   |
| **忽略 (Skip)**                  | **👍 安全、最快**。不處理這個檔案，繼續處理其他無衝突檔案。                             | 適用於使用者只是誤選了檔案，或確認現有表格已經是最新的。         |
| **取消上傳 (Cancel)**            | **👍 最保守**。停止所有上傳，讓使用者整理檔案。                                         | 適用於衝突數量過多，需要使用者手動調整檔名。                     |

## Gemini 回答 資料表多重上傳-檔名重複檢查
這是一個經典的 UX 與系統性能之間的權衡問題。對於您的 **Electron + better-sqlite3** 桌面應用程式來說，由於 **IPC 通訊成本低**（不需要經過網路），我強烈推薦採用 **「選擇檔案後立即檢查」** 的策略。

以下是針對兩種檢查時機的專業分析：

-----

### 檢查時機分析

#### 策略一：選擇檔案後**立即**向後端發送請求進行檢查 (Immediately)

當使用者在檔案選擇框（$\text{handleFileChange}$）或拖曳操作（$\text{handleDrop}$）完成後，前端立即將新的檔名列表通過 $\text{IPC}$ 傳遞給後端進行比對。

| 優點 (UX/系統)                                                                                | 缺點 (UX/系統)                                                                                                                           | 消耗 (時間/空間)                                                                          |
| :-------------------------------------------------------------------------------------------- | :--------------------------------------------------------------------------------------------------------------------------------------- | :---------------------------------------------------------------------------------------- |
| **即時反饋 (Instant Feedback)**：使用者在選擇檔案的**當下**就能看到哪些檔案有衝突，體驗極佳。 | **潛在的多次 $\text{IPC}$ 呼叫**：如果使用者多次添加或移除檔案，會觸發多次後端檢查。                                                     | **時間消耗：低**。$\text{better-sqlite3}$ 查詢所有表格名稱非常快。$\text{IPC}$ 延遲極低。 |
| **最佳化決策流程**：使用者可以在整理檔案時就解決衝突，減少最後一步的心理負擔。                | **前端狀態管理複雜度增加**：需要一個獨立的 $\text{useState}$ 或 $\text{Zustand}$ 狀態（如 $\text{filesStatus}$）來存儲檔案與其衝突狀態。 | **空間消耗：極低**。僅傳輸檔名列表，資料庫操作也只涉及查詢表格名稱的 $\text{metadata}$。  |
| **分攤延遲**：將檢查的微小延遲分散到使用者操作過程中，而不是集中在點擊「確認」時。            | **短暫的 UI 更新**：檢查結果返回前，衝突標記可能會有短暫的延遲出現。                                                                     |                                                                                           |
| **強制性檢查**：在 $\text{selectedFiles}$ 狀態更新的生命週期中執行，不易被忽略。              |                                                                                                                                          |                                                                                           |

-----

#### 策略二：等到使用者點擊「確認」上傳前再檢查 (On Confirmation)

在 $\text{handleUpload}$ 函式被呼叫時，在啟動實際的上傳流程前，先發送 $\text{IPC}$ 請求進行衝突檢查。

| 優點 (UX/系統)                                                         | 缺點 (UX/系統)                                                                                                                       | 消耗 (時間/空間)                                                           |
| :--------------------------------------------------------------------- | :----------------------------------------------------------------------------------------------------------------------------------- | :------------------------------------------------------------------------- |
| **最少 $\text{IPC}$ 呼叫**：只在點擊「確認」時檢查一次，減少通訊次數。 | **「確認」按鈕前的延遲**：如果檔案列表很長，點擊「確認」後會有一小段延遲（雖然很短，但在桌面應用中體驗仍顯突兀）。                   | **時間消耗：低**。與策略一的後端操作時間相同，但**全部集中**在點擊確認時。 |
| **前端狀態管理簡單**：狀態只需要在點擊確認時被處理。                   | **不佳的使用者體驗**：使用者可能點了「確認」後才發現有大量衝突，需要回頭處理，增加了摩擦。                                           | **空間消耗：極低**。與策略一相同。                                         |
|                                                                        | **流程中斷風險**：如果檢查發現衝突，需要彈出模態框或修改 $\text{DialogContent}$ 來讓使用者選擇處理方式，這會中斷「確認上傳」的意圖。 |                                                                            |

-----

### 專業建議與實作優化

對於您的專案，我強烈建議選擇 **策略一：選擇檔案後立即檢查**。

由於您使用的是 **Electron**，$\text{IPC}$ 通訊是**非同步且極快**的本地通訊，檢查資料庫中的表格名稱列表幾乎是**瞬間完成**的操作。將這個極短的延遲分散到使用者選擇檔案的過程中，可以換來最好的使用者體驗。

## Gemini 回答 useDebounce
當然！`useDebounce` 是在 React 應用程式中處理高頻率事件時的**性能優化利器**。

-----

### 什麼是 Debounce (去抖動)?

**去抖動**（Debounce）是一種編程技術，用於**限制函數被執行的頻率**。

想像一下你正在按電梯按鈕：如果你瘋狂連按 10 次，電梯只會響應**一次**。Debounce 的邏輯類似：它確保一個函數在特定時間間隔內，**只會在最後一次觸發後的一段延遲時間才真正執行**。

#### 為什麼使用 `useDebounce`？

在你的專案中，你希望在使用者選擇多個檔案時（高頻率事件，例如拖曳多個檔案、多次點擊新增），只在使用者**停止操作**一段時間（例如 300 毫秒）後，才觸發「向後端檢查檔案名稱衝突」的 $\text{IPC}$ 呼叫。

這能有效：

1.  **減少 $\text{IPC}$ 呼叫次數**：避免短時間內對 $\text{better-sqlite3}$ 進行不必要的重複查詢。
2.  **改善使用者體驗 (UX)**：確保後端檢查是在使用者操作穩定後才發生，提供最終的、穩定的衝突狀態。

### TypeScript 版本的 `useDebounce` 實作範例

`useDebounce` 通常作為一個自定義 Hook 來實作，它接收一個要監聽的 **`value`** 和一個 **`delay`** 毫秒數，並回傳一個只有在延遲結束後才會更新的 **`debouncedValue`**。

在 `src/hooks/` 資料夾下，你可以新增一個 `useDebounce.tsx` 檔案：

```tsx
// src/hooks/useDebounce.tsx
import { useState, useEffect } from 'react';

/**
 * useDebounce Hook
 *
 * 在指定的延遲時間內，限制 value 的更新頻率。
 * 只有當 value 在 delay 毫秒內沒有變化時，才會更新 debouncedValue。
 *
 * @param value 任何類型的值 (例如字串、陣列)，當它改變時會觸發 debounce 計時器。
 * @param delay 延遲時間，單位為毫秒 (ms)。
 * @returns 經過 debounce 處理後的值。
 */
export function useDebounce<T>(value: T, delay: number): T {
  // 1. 儲存經過延遲後才會更新的值
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    // 2. 設定一個新的計時器
    const handler = setTimeout(() => {
      // 延遲時間到後，將最新的 value 設定給 debouncedValue
      setDebouncedValue(value);
    }, delay);

    // 3. 清理函數 (Cleanup Function)
    // 在下一次 useEffect 執行前（即 value 再次改變時）或元件卸載時，
    // 清除上一次設定的計時器。
    // 這是 debounce 邏輯的核心：如果計時器還沒跑完 value 又變了，就重設。
    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]); // 依賴於傳入的 value 和 delay

  return debouncedValue;
}
```

#### 如何在 `UploadDataTableDialog.tsx` 中使用

有了這個 Hook，你就能像上一個回答建議的那樣，在 `UploadDataTableDialog.tsx` 中將\*\*「選中的檔案列表」\*\*進行去抖動處理，然後監聽去抖動後的結果。

```tsx
// src/components/DataTablesPage/UploadDataTableDialog.tsx (部分修改)
import { useDebounce } from '../../hooks/useDebounce'; // 假設你已匯入

// ... (其他 state 和函式)

// 1. 使用 useDebounce 處理 selectedFiles 狀態
const debouncedSelectedFiles = useDebounce(selectedFiles, 300); // 設置 300ms 延遲

// 2. 監聽 debouncedSelectedFiles 的變化，發起 IPC 呼叫
useEffect(() => {
  // 只有當 debouncedSelectedFiles 有內容時才執行檢查
  if (debouncedSelectedFiles.length > 0) {
    const fileNames = debouncedSelectedFiles.map(f => f.name);
    // 這是觸發後端檢查的 IPC 呼叫，它只會在使用者停止選擇檔案 300ms 後才執行
    window.electron.ipcRenderer.invoke('check-table-conflict', fileNames)
      .then((conflictResult) => {
        // ... 更新 filesStatus 狀態來渲染衝突圖標和選項
        // setFilesStatus(...)
      });
  } else {
    // 清空 filesStatus
    // setFilesStatus([]); 
  }
}, [debouncedSelectedFiles]); // 監聽 debouncedSelectedFiles
```

透過這種方式，即使使用者連續拖曳或選擇多個檔案，`check-table-conflict` 這個 $\text{IPC}$ 呼叫也只會被觸發**一次**，從而實現高效、流暢的即時衝突檢查。
